% -----------------------------------------------------------------------------
%  Answer space

\newlength{\wsgLineSep}      \setlength{\wsgLineSep}{9mm}
\newlength{\wsgGridStep}     \setlength{\wsgGridStep}{5mm}
\newcounter{wsgLineCounter}
\newcount\wsgGridCols
\newcount\wsgGridRows

% \wsgAnswerLines{n} -- n ruled lines
\newcommand{\wsgAnswerLines}[1]{%
  \par\vspace{0.6em}%
  \setcounter{wsgLineCounter}{0}%
  \loop\ifnum\value{wsgLineCounter}<#1\relax
    \noindent\textcolor{wsgRule}{\rule{\linewidth}{0.4pt}}\par
    \vspace{\wsgLineSep}%
    \stepcounter{wsgLineCounter}%
  \repeat
  \vspace{-\wsgLineSep}\par\vspace{0.3em}%
}

% \wsgAnswerDots{n} -- n dotted lines
\newcommand{\wsgAnswerDots}[1]{%
  \par\vspace{0.6em}%
  \setcounter{wsgLineCounter}{0}%
  \loop\ifnum\value{wsgLineCounter}<#1\relax
    \noindent\textcolor{wsgRule}{\makebox[\linewidth]{\dotfill}}\par
    \vspace{\wsgLineSep}%
    \stepcounter{wsgLineCounter}%
  \repeat
  \vspace{-\wsgLineSep}\par\vspace{0.3em}%
}

% \wsgAnswerBlank{height} -- plain vertical space
\newcommand{\wsgAnswerBlank}[1]{\par\vspace{#1}\par}

% \wsgAnswerBox{height} -- empty framed box
\newcommand{\wsgAnswerBox}[1]{%
  \par\vspace{0.6em}%
  \noindent\textcolor{wsgRule}{\framebox[\linewidth]{\rule{0pt}{#1}}}%
  \par\vspace{0.3em}%
}

% \wsgAnswerGrid{height} -- framed box with a 5 mm square grid
\newcommand{\wsgAnswerGrid}[1]{%
  \par\vspace{0.6em}%
  \begingroup
  \wsgGridCols=\numexpr\linewidth/\wsgGridStep\relax
  \wsgGridRows=\numexpr\dimexpr#1\relax/\wsgGridStep\relax
  \setlength{\unitlength}{\wsgGridStep}%
  \noindent
  \begin{picture}(\the\wsgGridCols,\the\wsgGridRows)
    \textcolor{wsgRule!45}{%
      \multiput(0,0)(1,0){\numexpr\wsgGridCols+1\relax}{\line(0,1){\the\wsgGridRows}}%
      \multiput(0,0)(0,1){\numexpr\wsgGridRows+1\relax}{\line(1,0){\the\wsgGridCols}}%
    }%
  \end{picture}%
  \endgroup
  \par\vspace{0.3em}%
}

% -----------------------------------------------------------------------------
%  Task environment
%
%    \begin{task}[title=..., points=3, stars=1, bloom=..., answer=lines, lines=5]
%       statement of the task
%       \begin{solution}
%          solution, printed in the solution sheet only
%       \end{solution}
%    \end{task}
%
%  answer = none | blank | lines | dots | box | grid   (default: none)
%    lines=<n>    number of lines for answer=lines / answer=dots
%    space=<len>  height for answer=blank / answer=box / answer=grid
%    stars=<n>    difficulty, printed as n asterisks behind the number; any
%                 other value is printed as it stands (stars=+ works too)
%    bloom=<text> note for the teacher, printed in the solution sheet only
% -----------------------------------------------------------------------------
\newcounter{wsgTaskCounter}
\newcommand{\wsgPointSum}{0}
\providecommand{\wsgTotalPoints}{?}
\providecommand{\wsgAnyStars}{false}
\newcommand{\wsgStarsSeen}{false}

% \wsgStarMark{1} -> *, \wsgStarMark{2} -> **, anything else is printed as is
\newcommand{\wsgStarMark}[1]{%
  \textsuperscript{%
    \ifstrequal{#1}{1}{*}{\ifstrequal{#1}{2}{**}{#1}}%
  }%
}

\makeatletter
\define@key{wsgtask}{title}{\def\wsg@task@title{#1}}
\define@key{wsgtask}{points}{\def\wsg@task@points{#1}}
\define@key{wsgtask}{answer}{\def\wsg@task@answer{#1}}
\define@key{wsgtask}{lines}{\def\wsg@task@lines{#1}}
\define@key{wsgtask}{space}{\def\wsg@task@space{#1}}
\define@key{wsgtask}{stars}{\def\wsg@task@stars{#1}}
\define@key{wsgtask}{bloom}{\def\wsg@task@bloom{#1}}

\newcommand{\wsg@answerspace}{%
  \ifdefstring{\wsg@task@answer}{blank}{\wsgAnswerBlank{\wsg@task@space}}{}%
  \ifdefstring{\wsg@task@answer}{lines}{\wsgAnswerLines{\wsg@task@lines}}{}%
  \ifdefstring{\wsg@task@answer}{dots}{\wsgAnswerDots{\wsg@task@lines}}{}%
  \ifdefstring{\wsg@task@answer}{box}{\wsgAnswerBox{\wsg@task@space}}{}%
  \ifdefstring{\wsg@task@answer}{grid}{\wsgAnswerGrid{\wsg@task@space}}{}%
}

\newenvironment{task}[1][]{%
  \def\wsg@task@title{}%
  \def\wsg@task@points{}%
  \def\wsg@task@answer{none}%
  \def\wsg@task@lines{4}%
  \def\wsg@task@space{4cm}%
  \def\wsg@task@stars{}%
  \def\wsg@task@bloom{}%
  \setkeys{wsgtask}{#1}%
  \stepcounter{wsgTaskCounter}%
  \ifdefempty{\wsg@task@points}{}{%
    \xdef\wsgPointSum{\fpeval{\wsgPointSum+\wsg@task@points}}%
  }%
  \ifdefempty{\wsg@task@stars}{}{\gdef\wsgStarsSeen{true}}%
  % Right hand side of the title line: the points and, in the solution sheet,
  % the note for the teacher.
  \def\wsg@task@rhs{}%
  \ifwsgGraded
    \ifdefempty{\wsg@task@points}{}{%
      \def\wsg@task@rhs{(\wsg@task@points\,\wsgLabelPointsShort)}}%
  \fi
  \ifwsgSolutions
    \ifdefempty{\wsg@task@bloom}{}{%
      \ifdefempty{\wsg@task@rhs}{}{\appto\wsg@task@rhs{\quad}}%
      \appto\wsg@task@rhs{\wsgLabelBloom: \wsg@task@bloom}}%
  \fi
  \par\addvspace{1.1em}\needspace{3\baselineskip}%
  \noindent
  {\bfseries\color{wsgAccent}\wsgLabelTask~\thewsgTaskCounter
    \ifdefempty{\wsg@task@stars}{}{%
      \expandafter\wsgStarMark\expandafter{\wsg@task@stars}}%
    \ifdefempty{\wsg@task@title}{}{\,--\, \wsg@task@title}}%
  \ifdefempty{\wsg@task@rhs}{}{%
    \hfill{\small\color{wsgMuted}\wsg@task@rhs}}%
  \par\nobreak\vspace{0.35em}%
  \noindent\ignorespaces
}{%
  \ifwsgSolutions\else\wsg@answerspace\fi
  \par\addvspace{0.4em}%
}
\makeatother

% -----------------------------------------------------------------------------
%  Solution environment
%
%  In the solution sheet the content is printed, in the assignment sheet it is
%  still processed -- so that listings and verbatim material keep working --
%  but the result is thrown away instead of being placed on the page.
% -----------------------------------------------------------------------------
\newsavebox{\wsgDiscardBox}

\newcommand{\wsgSolutionStart}{%
  \par\vspace{0.5em}%
  \begingroup\color{wsgSolution}%
  \noindent{\bfseries\wsgLabelSolution:}\par\nobreak\vspace{0.2em}%
  \noindent\ignorespaces
}
\newcommand{\wsgSolutionEnd}{\par\endgroup\vspace{0.3em}}

\newenvironment{solution}{%
  \ifwsgSolutions
    \wsgSolutionStart
  \else
    \setbox\wsgDiscardBox=\vbox\bgroup
    \hbadness=10000 \vbadness=10000 \hfuzz=\maxdimen \vfuzz=\maxdimen
    \ignorespaces
  \fi
}{%
  \ifwsgSolutions
    \wsgSolutionEnd
  \else
    \par\egroup
  \fi
}

% -----------------------------------------------------------------------------
%  Values collected during the previous compilation run: the total number of
%  points printed in the header and whether any task carries a difficulty mark.
% -----------------------------------------------------------------------------
\makeatletter
\newcommand{\wsgSetTotal}[1]{\gdef\wsgTotalPoints{#1}}
\newcommand{\wsgSetStars}[1]{\gdef\wsgAnyStars{#1}}
\AtEndDocument{%
  \if@filesw
    \immediate\write\@auxout{\string\wsgSetTotal{\wsgPointSum}}%
    \immediate\write\@auxout{\string\wsgSetStars{\wsgStarsSeen}}%
  \fi
  \ifx\wsgTotalPoints\wsgPointSum\else
    \PackageWarningNoLine{wsg}{Points changed. Rerun to get the total right}%
  \fi
  \ifx\wsgAnyStars\wsgStarsSeen\else
    \PackageWarningNoLine{wsg}{Difficulty marks changed. Rerun the compilation}%
  \fi
}
\makeatother

% Headings used to group the tasks of a worksheet into blocks. \needspace keeps
% a heading from being left alone at the bottom of a page -- the task that
% follows starts with a page break of its own, so LaTeX cannot prevent it.

\newcommand{\wsgMakeHeader}{%
  \noindent\textcolor{wsgAccent}{\rule{\linewidth}{1.2pt}}\par\vspace{0.3em}%
  \noindent
  \begin{minipage}[b]{0.62\linewidth}
    \ifdefempty{\wsgSchool}{}{{\footnotesize\color{wsgMuted}\wsgSchool}\par}%
    {\large\bfseries\color{wsgAccent}\wsgSubject}%
  \end{minipage}%
  \hfill
  \begin{minipage}[b]{0.36\linewidth}
    \raggedleft\footnotesize\color{wsgMuted}%
    \ifdefempty{\wsgClass}{}{\wsgClass\par}%
    \ifdefempty{\wsgYear}{}{\wsgYear}%
  \end{minipage}%
  \par\vspace{0.5em}%
  \noindent\colorbox{wsgLight}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \vspace{0.35em}%
      {\large\bfseries\wsgLabelWorksheet\ \wsgLabelNo~\wsgNumber}%
      \ifdefempty{\wsgTitle}{}{\ \textbar\ {\large\wsgTitle}}%
      \ifwsgSolutions\hfill\textcolor{wsgSolution}{\bfseries\footnotesize\wsgLabelSolutionSheet}\fi
      \vspace{0.35em}%
    \end{minipage}}%
  \par\vspace{0.6em}%
  \ifwsgSolutions\else
    \ifboolexpr{bool{wsgCredentials} or bool{wsgGraded}}{%
      \noindent
      \ifwsgCredentials
        \wsgLabelName:~\textcolor{wsgRule}{\rule[-0.35em]{0.45\linewidth}{0.4pt}}%
      \fi
      \ifwsgGraded
        \hfill\wsgLabelPoints:~\textcolor{wsgRule}{\rule[-0.35em]{1.6cm}{0.4pt}}\,/\,\wsgTotalPoints
      \fi
      \par\vspace{0.4em}%
    }{}%
  \fi
  <<GRADING_TABLE>>%
  \ifdefstring{\wsgAnyStars}{true}{%
    \noindent{\footnotesize\color{wsgMuted}\wsgLabelStarNote}\par\vspace{0.4em}%
  }{}%
  \noindent\textcolor{wsgRule}{\rule{\linewidth}{0.4pt}}\par\vspace{0.8em}%
}
