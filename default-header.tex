\newlength{\wsgLineSep}      \setlength{\wsgLineSep}{9mm}

% \wsgAnswerLines{n} -- n ruled lines
\newcommand{\wsgAnswerLines}[1]{%
  \par\vspace{0.6em}%
  \setcounter{wsgLineCounter}{0}%
  \loop\ifnum\value{wsgLineCounter}<#1\relax
    \noindent\textcolor{wsgRule}{\rule{\linewidth}{0.4pt}}\par
    \vspace{\wsgLineSep}%
    \stepcounter{wsgLineCounter}%
  \repeat
  \vspace{-\wsgLineSep}\par\vspace{0.3em}%
}

% \wsgAnswerDots{n} -- n dotted lines
\newcommand{\wsgAnswerDots}[1]{%
  \par\vspace{0.6em}%
  \setcounter{wsgLineCounter}{0}%
  \loop\ifnum\value{wsgLineCounter}<#1\relax
    \noindent\textcolor{wsgRule}{\makebox[\linewidth]{\dotfill}}\par
    \vspace{\wsgLineSep}%
    \stepcounter{wsgLineCounter}%
  \repeat
  \vspace{-\wsgLineSep}\par\vspace{0.3em}%
}

% \wsgAnswerBlank{height} -- plain vertical space
\newcommand{\wsgAnswerBlank}[1]{\par\vspace{#1}\par}

% \wsgAnswerBox{height} -- empty framed box
\newcommand{\wsgAnswerBox}[1]{%
  \par\vspace{0.6em}%
  \noindent\textcolor{wsgRule}{\framebox[\linewidth]{\rule{0pt}{#1}}}%
  \par\vspace{0.3em}%
}

% \wsgAnswerGrid{height} -- framed box with a 5 mm square grid
\newcommand{\wsgAnswerGrid}[1]{%
  \par\vspace{0.6em}%
  \begingroup
  \wsgGridCols=\numexpr\linewidth/\wsgGridStep\relax
  \wsgGridRows=\numexpr\dimexpr#1\relax/\wsgGridStep\relax
  \setlength{\unitlength}{\wsgGridStep}%
  \noindent
  \begin{picture}(\the\wsgGridCols,\the\wsgGridRows)
    \textcolor{wsgRule!45}{%
      \multiput(0,0)(1,0){\numexpr\wsgGridCols+1\relax}{\line(0,1){\the\wsgGridRows}}%
      \multiput(0,0)(0,1){\numexpr\wsgGridRows+1\relax}{\line(1,0){\the\wsgGridCols}}%
    }%
  \end{picture}%
  \endgroup
  \par\vspace{0.3em}%
}

% -----------------------------------------------------------------------------
%  Task environment
%
%    \begin{task}[title=..., points=3, stars=1, answer=lines, lines=5]
%       statement of the task
%       \begin{solution}
%          solution, printed in the solution sheet only
%       \end{solution}
%    \end{task}
%
%  answer = none | blank | lines | dots | box | grid   (default: none)
%    the answer space is left out in the solution sheet and whenever the
%    build was started with --no-answer-space
%    lines=<n>    number of lines for answer=lines / answer=dots
%    space=<len>  height for answer=blank / answer=box / answer=grid
%    stars=<n>    difficulty, printed as n asterisks behind the number; any
%                 other value is printed as it stands (stars=+ works too)
% -----------------------------------------------------------------------------
\newcounter{wsgTaskCounter}
% Starred, i.e. not \long: at the end of the document they are compared by
% \ifx with the values read back from the .aux file, which are not \long
% either -- otherwise a sheet without points or stars warns on every run.
\newcommand*{\wsgPointSum}{0}
\providecommand{\wsgTotalPoints}{?}
\providecommand{\wsgAnyStars}{false}
\newcommand*{\wsgStarsSeen}{false}

% \wsgStarMark{1} -> *, \wsgStarMark{2} -> **, anything else is printed as is
\newcommand{\wsgStarMark}[1]{%
  \textsuperscript{%
    \ifstrequal{#1}{1}{*}{\ifstrequal{#1}{2}{**}{#1}}%
  }%
}

\newcommand{\wsg@answerspace}{%
  \ifdefstring{\wsg@task@answer}{blank}{\wsgAnswerBlank{\wsg@task@space}}{}%
  \ifdefstring{\wsg@task@answer}{lines}{\wsgAnswerLines{\wsg@task@lines}}{}%
  \ifdefstring{\wsg@task@answer}{dots}{\wsgAnswerDots{\wsg@task@lines}}{}%
  \ifdefstring{\wsg@task@answer}{box}{\wsgAnswerBox{\wsg@task@space}}{}%
  \ifdefstring{\wsg@task@answer}{grid}{\wsgAnswerGrid{\wsg@task@space}}{}%
}

% The answer space of the current task is printed at most once: at the end of
% the task, or earlier, when a results box inside the task asks for it so that
% the results come after the room for the answer.
\newif\ifwsg@answerpending
\newcommand{\wsg@flushanswerspace}{%
  \ifwsg@answerpending
    \global\wsg@answerpendingfalse
    \ifwsgSolutions\else\ifwsgAnswerSpace\wsg@answerspace\fi\fi
  \fi
}

% -----------------------------------------------------------------------------
%  Shared bookkeeping of one task
% -----------------------------------------------------------------------------
% Reads the options, advances the counter, adds up the points and builds
% \wsg@task@rhs, i.e. the right hand side of the title line.
\newcommand{\wsgTaskSetup}[1]{%
  \def\wsg@task@title{}%
  \def\wsg@task@points{}%
  \def\wsg@task@answer{none}%
  \def\wsg@task@lines{4}%
  \def\wsg@task@space{4cm}%
  \def\wsg@task@stars{}%
  \setkeys{wsgtask}{#1}%
  \stepcounter{wsgTaskCounter}%
  \global\wsg@answerpendingtrue
  \ifdefempty{\wsg@task@points}{}{%
    \xdef\wsgPointSum{\fpeval{\wsgPointSum+\wsg@task@points}}%
  }%
  \ifdefempty{\wsg@task@stars}{}{\gdef\wsgStarsSeen{true}}%
  \def\wsg@task@rhs{}%
  \ifwsgGraded
    \ifdefempty{\wsg@task@points}{}{%
      \def\wsg@task@rhs{\wsg@task@points\,\wsgLabelPointsShort}}%
  \fi
}

% Difficulty marks of the current task, empty when the task carries none.
\newcommand{\wsgTaskStars}{%
  \ifdefempty{\wsg@task@stars}{}{%
    \expandafter\wsgStarMark\expandafter{\wsg@task@stars}}%
}

% Title of the current task, the fallback is used when no title was given.
\newcommand{\wsgTaskTitle}[1]{%
  \ifdefempty{\wsg@task@title}{#1}{\wsg@task@title}%
}

% -----------------------------------------------------------------------------
%  Task environment -- large two digit number, uppercase title
% -----------------------------------------------------------------------------
\newcommand{\wsgTwoDigits}{%
  \ifnum\value{wsgTaskCounter}<10 0\fi\thewsgTaskCounter}

\newenvironment{task}[1][]{%
  \wsgTaskSetup{#1}%
  \par\addvspace{2.2em}\needspace{4\baselineskip}%
  \noindent
  {\fontsize{15}{15}\selectfont\bfseries\color{wsgAccent}\wsgTwoDigits}%
  \hspace{0.8em}%
  \ifdefempty{\wsg@task@title}{}{%
    {\bfseries\MakeUppercase{\wsg@task@title}}}%
  {\color{wsgSolution}\wsgTaskStars}%
  \ifdefempty{\wsg@task@rhs}{}{%
    \hfill{\footnotesize\color{wsgMuted}\MakeUppercase{\wsg@task@rhs}}}%
  \par\nobreak\vspace{0.5em}%
  \noindent\ignorespaces
}{%
  \wsg@flushanswerspace
  \par\addvspace{0.6em}%
}
\makeatother

% -----------------------------------------------------------------------------
%  Reminder, note and results -- yellow field, thin frame
%
%    \begin{reminder}[title] ... \end{reminder}  definition, theorem or method
%                                                recalled before the tasks
%    \begin{note}[title] ... \end{note}          remark for the students
%    \begin{results}[title] ... \end{results}    short results of a task
%
%  The optional title is printed behind the label. All three boxes appear in
%  both sheets; --no-results leaves the results out of the assignment sheet.
%  Inside the results box every enumerate is set inline, so the results of
%  the subtasks run on like a paragraph instead of taking a line each. Within
%  a task with an answer space, the space is printed before the results.
%  The boxes are short and never split -- a box that does not fit moves to
%  the next page as a whole.
% -----------------------------------------------------------------------------
\tcbset{
  wsgbox/.style={
    enhanced, sharp corners, parbox=false,
    left=3mm, right=3mm, top=2mm, bottom=2mm,
    fontupper=\small
  },
  wsgnote/.style={
    wsgbox, frame hidden, colback=wsgNoteBg,
    borderline west={2.4pt}{0pt}{wsgNoteBar},
    before skip=1.2em, after skip=1.2em
  },
  wsgresults/.style={
    wsgbox, colback=white, colframe=wsgRule, boxrule=0.6pt,
    before skip=0.8em, after skip=0.8em
  }
}

% -----------------------------------------------------------------------------
%  Values collected during the previous compilation run: the total number of
%  points printed in the header and whether any task carries a difficulty mark.
% -----------------------------------------------------------------------------
\makeatletter
\newcommand{\wsgSetTotal}[1]{\gdef\wsgTotalPoints{#1}}
\newcommand{\wsgSetStars}[1]{\gdef\wsgAnyStars{#1}}
\AtEndDocument{%
  \if@filesw
    \immediate\write\@auxout{\string\wsgSetTotal{\wsgPointSum}}%
    \immediate\write\@auxout{\string\wsgSetStars{\wsgStarsSeen}}%
  \fi
  \ifx\wsgTotalPoints\wsgPointSum\else
    \PackageWarningNoLine{wsg}{Points changed. Rerun to get the total right}%
  \fi
  \ifx\wsgAnyStars\wsgStarsSeen\else
    \PackageWarningNoLine{wsg}{Difficulty marks changed. Rerun the compilation}%
  \fi
}
\makeatother
