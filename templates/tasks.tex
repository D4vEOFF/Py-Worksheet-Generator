\begin{reminder}[Cyklus for]
  Cyklus \texttt{for} provede tělo cyklu jednou pro každý prvek sekvence,
  \texttt{range(n)} vytvoří sekvenci čísel $0, 1, \dots, n-1$.
\end{reminder}

\begin{task}[title=Zahřívací otázka, points=2, answer=lines, lines=3]
  Vysvětlete vlastními slovy, k čemu slouží cyklus \texttt{for}.

  \begin{solution}
    Slouží k opakovanému provedení bloku příkazů pro každý prvek sekvence.
  \end{solution}
\end{task}

\begin{task}[title=Rozbor kódu, points=3, answer=dots, lines=2]
  Co vypíší následující programy?
  \begin{enumerate}
    \item \verb|for i in range(3): print(i * i)|
    \item \verb|for c in "abc": print(c.upper())|
  \end{enumerate}

  \begin{results}
    \begin{enumerate}
      \item 0, 1, 4;
      \item A, B, C.
    \end{enumerate}
  \end{results}

  \begin{solution}
    Každý program vypíše tři řádky: nejprve druhé mocniny čísel 0, 1 a 2,
    potom písmena řetězce převedená na velká.
  \end{solution}
\end{task}

\begin{task}[points=5, stars=1, answer=box, space=5cm]
  Napište program, který vypíše všechna sudá čísla od 1 do 20.

  \begin{solution}
    \begin{code}[Python]
for i in range(2, 21, 2):
    print(i)
\end{code}
  \end{solution}
\end{task}
